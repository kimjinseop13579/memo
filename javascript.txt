자바 스크립트 함수
자바 스크립트에는 타입이 따로 정해져 있지않아서 정수, 소수, 문자열 모두 들어갈수있다
num = 10;
num = 3.14;
num = "문자열";

 - var 를 변수앞에 붙여주면 재선언 할수있음 빨간줄이 뜨지않음
 -- var num = 13;
 -- num = 20;
 -- var num = 50;
 헷갈려서 잘 안씀

 - let 한번 선언한 변수는 다시 선언못함
 -- let a = 40;
 -- a = 400;
    이후에 같은 이름의 변수로 다시 선언하면 빨간줄뜸

 - const 내부데이터를 변환할수 없음 변환 하려고하면 빨간줄은 안뜸
 --  const b = 40
 --  b = 50;
   빨간줄이 따로 뜨지는 않지만 오류는 뜸

 - 자바스크립트에서 배열은 자바의 리스트처럼 사용하면 됨  
string_word = "문자열만";
      int_num = 10;
    let arr = new Array();
     const arr2 = [10,20,30];
